Absolutely! tmux is a super useful terminal multiplexer that lets you run multiple terminal sessions inside one window, detach them, and come back later—perfect for managing long-running programs or organizing work in the terminal. I’ll give you a clear step-by-step guide.

1. Installing tmux

On most systems:

macOS (with Homebrew):
brew install tmux
Ubuntu / Debian:
sudo apt update
sudo apt install tmux
Fedora / CentOS:
sudo dnf install tmux

Check if it installed correctly:

tmux -V

You should see something like tmux 3.4.

2. Starting tmux

Just type:

tmux

You’ll enter a new session. The bottom line shows your status (current window, time, etc.).

3. Basic tmux commands

tmux commands are usually prefixed with Ctrl + b (we call this the "prefix"). After pressing Ctrl + b, release and then press the next key.

Windows and Panes
Create a new window: Ctrl+b c
Switch windows: Ctrl+b n (next) or Ctrl+b p (previous)
List windows: Ctrl+b w
Rename a window: Ctrl+b ,
Split pane horizontally: Ctrl+b "
Split pane vertically: Ctrl+b %
Switch panes: Ctrl+b + arrow key
Resize panes: Ctrl+b + hold Ctrl + arrow keys
Close a pane: exit or Ctrl+b x
Session Management
List sessions: tmux ls
Detach from session: Ctrl+b d (you can leave it running)
Attach to a session:
tmux attach -t session_name
Kill a session:
tmux kill-session -t session_name

Tip: If you want to start a session with a name:

tmux new -s mysession
Copy & Paste
Enter copy mode: Ctrl+b [
Move cursor with arrow keys.
Press Space to start selection, move to end, then Enter to copy.
Paste with: Ctrl+b ]
Customizing tmux

You can create a config file ~/.tmux.conf for settings. Example:

# Use vim-like pane switching
setw -g mode-keys vi

# Set prefix to Ctrl+a instead of Ctrl+b
unbind C-b
set-option -g prefix C-a
bind-key C-a send-prefix

# Enable mouse support
set -g mouse on

Then reload config in tmux:

Ctrl+b :source-file ~/.tmux.conf
Quick Workflow Example
Start a session: tmux new -s dev
Split window vertically: Ctrl+b %
Run your server in one pane, editor in the other
Detach: Ctrl+b d
Reattach later: tmux attach -t dev

What that means

A steering model should either:

naturally stay inside [-1, 1] because of a tanh output, or
be clamped before use.

If it is producing values outside that range, then one of these is happening:

the inference code is using the raw output before the final activation,
the final activation is missing or bypassed,
or the model was trained on targets that were not truly normalized.

import os
import json
import argparse
from glob import glob

import cv2
import numpy as np
import torch
import torch.nn as nn
import torch.optim as optim
from torch.utils.data import Dataset, DataLoader, random_split

torch.backends.cudnn.benchmark = True

DEVICE = "cuda" if torch.cuda.is_available() else "cpu"


def clamp(x, lo=-1.0, hi=1.0):
    return float(max(lo, min(hi, float(x))))


def discover_roots(explicit_roots=None):
    if explicit_roots:
        roots = [r for r in explicit_roots if os.path.isdir(r)]
        if roots:
            return roots

    preferred = [
        "dataset_l2_realistic",
        "dataset_l2_balanced",
        "dataset_realistic",
        "dataset_balanced",
    ]

    roots = [r for r in preferred if os.path.isdir(r) and os.path.isfile(os.path.join(r, "labels.json"))]
    if roots:
        return roots

    candidates = []
    for p in glob("dataset*"):
        if os.path.isdir(p) and os.path.isfile(os.path.join(p, "labels.json")):
            candidates.append(p)

    if candidates:
        return sorted(candidates)

    raise FileNotFoundError(
        "No dataset roots found. Expected folders like dataset_l2_realistic or dataset_realistic "
        "with labels.json inside."
    )


def preprocess_image(img):
    img = cv2.resize(img, (160, 120))
    img = img.astype(np.float32) / 255.0
    img = (img - 0.5) / 0.5
    img = np.transpose(img, (2, 0, 1))
    return torch.from_numpy(img).float()


class SteeringDataset(Dataset):
    def __init__(self, roots):
        self.samples = []
        self.targets = []

        for root in roots:
            label_path = os.path.join(root, "labels.json")
            with open(label_path, "r") as f:
                items = json.load(f)

            for item in items:
                img_path = os.path.join(root, "images", item["image"])
                if not os.path.exists(img_path):
                    continue

                steer = clamp(item.get("steering", 0.0))
                self.samples.append((img_path, steer))
                self.targets.append(steer)

        if not self.samples:
            raise FileNotFoundError("No usable images found in the provided dataset roots.")

        t = np.array(self.targets, dtype=np.float32)
        print("Loaded samples:", len(self.samples))
        print(f"Target range: min={t.min():.6f} max={t.max():.6f} mean={t.mean():.6f} std={t.std():.6f}")

    def __len__(self):
        return len(self.samples)

    def __getitem__(self, idx):
        img_path, steer = self.samples[idx]
        img = cv2.imread(img_path)
        if img is None:
            raise FileNotFoundError(img_path)

        img = preprocess_image(img)
        steer = torch.tensor([clamp(steer)], dtype=torch.float32)
        return img, steer


class SteeringNet(nn.Module):
    def __init__(self):
        super().__init__()

        self.features = nn.Sequential(
            nn.Conv2d(3, 24, 5, stride=2, padding=2),
            nn.BatchNorm2d(24),
            nn.ReLU(inplace=True),

            nn.Conv2d(24, 36, 5, stride=2, padding=2),
            nn.BatchNorm2d(36),
            nn.ReLU(inplace=True),

            nn.Conv2d(36, 48, 3, stride=2, padding=1),
            nn.BatchNorm2d(48),
            nn.ReLU(inplace=True),

            nn.Conv2d(48, 64, 3, stride=2, padding=1),
            nn.BatchNorm2d(64),
            nn.ReLU(inplace=True),

            nn.AdaptiveAvgPool2d((1, 1)),
        )

        self.head = nn.Sequential(
            nn.Flatten(),
            nn.Linear(64, 32),
            nn.ReLU(inplace=True),
            nn.Linear(32, 1),
            nn.Tanh(),
        )

    def forward(self, x):
        x = self.features(x)
        x = self.head(x)
        return x


def build_loader(dataset, batch_size, num_workers, shuffle, drop_last):
    kwargs = dict(
        batch_size=batch_size,
        shuffle=shuffle,
        num_workers=num_workers,
        pin_memory=True,
        drop_last=drop_last,
    )
    if num_workers > 0:
        kwargs["persistent_workers"] = True
        kwargs["prefetch_factor"] = 2
    return DataLoader(dataset, **kwargs)


def train(roots, epochs=20, batch_size=256, num_workers=8, lr=1e-4):
    dataset = SteeringDataset(roots)

    n = len(dataset)
    val_size = max(1, int(0.1 * n))
    train_size = max(1, n - val_size)

    gen = torch.Generator().manual_seed(42)
    train_ds, val_ds = random_split(dataset, [train_size, val_size], generator=gen)

    train_loader = build_loader(train_ds, batch_size, num_workers, True, True)
    val_loader = build_loader(val_ds, batch_size, max(0, num_workers // 2), False, False)

    model = SteeringNet().to(DEVICE)
    optimizer = optim.Adam(model.parameters(), lr=lr)
    loss_fn = nn.SmoothL1Loss(reduction="none")

    best_val = float("inf")
    loss_ema = None

    print("Device:", DEVICE)
    print("Roots:", roots)
    print(f"Train samples: {len(train_ds)} | Val samples: {len(val_ds)}")

    for epoch in range(epochs):
        model.train()
        train_total = 0.0

        for step, (imgs, targets) in enumerate(train_loader):
            imgs = imgs.to(DEVICE, non_blocking=True)
            targets = targets.to(DEVICE, non_blocking=True)

            preds = model(imgs)
            preds = torch.clamp(preds, -1.0, 1.0)

            raw = loss_fn(preds, targets)
            weights = 1.0 + 1.5 * torch.abs(targets)
            loss = (raw * weights).mean()

            optimizer.zero_grad(set_to_none=True)
            loss.backward()
            torch.nn.utils.clip_grad_norm_(model.parameters(), 1.0)
            optimizer.step()

            val = loss.item()
            train_total += val

            if loss_ema is None:
                loss_ema = val
            else:
                loss_ema = 0.95 * loss_ema + 0.05 * val

            if step % 50 == 0:
                print(f"Epoch {epoch} Step {step} Loss {val:.6f} EMA {loss_ema:.6f}")

        model.eval()
        val_total = 0.0
        mae_total = 0.0
        pred_min = 1e9
        pred_max = -1e9
        count = 0

        with torch.no_grad():
            for imgs, targets in val_loader:
                imgs = imgs.to(DEVICE, non_blocking=True)
                targets = targets.to(DEVICE, non_blocking=True)

                preds = model(imgs)
                preds = torch.clamp(preds, -1.0, 1.0)

                raw = loss_fn(preds, targets)
                weights = 1.0 + 1.5 * torch.abs(targets)
                vloss = (raw * weights).mean()

                val_total += vloss.item()
                mae_total += torch.mean(torch.abs(preds - targets)).item()

                pred_min = min(pred_min, float(preds.min().item()))
                pred_max = max(pred_max, float(preds.max().item()))
                count += 1

        avg_train = train_total / max(1, len(train_loader))
        avg_val = val_total / max(1, len(val_loader))
        avg_mae = mae_total / max(1, count)

        print(
            f"Epoch {epoch} DONE | "
            f"Train {avg_train:.6f} | Val {avg_val:.6f} | "
            f"MAE {avg_mae:.6f} | PredRange [{pred_min:.6f}, {pred_max:.6f}]"
        )

        if avg_val < best_val:
            best_val = avg_val
            torch.save(model.state_dict(), "steering_l2_best.pth")

    torch.save(model.state_dict(), "steering_l2.pth")
    print("Saved: steering_l2.pth")
    print("Saved best: steering_l2_best.pth")


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("--epochs", type=int, default=20)
    parser.add_argument("--batch-size", type=int, default=256)
    parser.add_argument("--workers", type=int, default=8)
    parser.add_argument("--lr", type=float, default=1e-4)
    parser.add_argument("--roots", nargs="*", default=None)
    args = parser.parse_args()

    roots = discover_roots(args.roots)
    train(
        roots=roots,
        epochs=args.epochs,
        batch_size=args.batch_size,
        num_workers=args.workers,
        lr=args.lr,
    )


if __name__ == "__main__":
    main()
